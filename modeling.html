calc_auc <- function(truth, estimate) {
  auc_obj <- pROC::roc(as.numeric(truth), as.numeric(estimate), quiet = TRUE)
  as.numeric(auc_obj$auc)
}

impute_numeric_median <- function(df) {
  df %>% mutate(across(where(is.numeric), ~ if_else(is.na(.), median(., na.rm = TRUE), .)))
}

Compare Logistic Regression, Naive Bayes, and Random Forest with 3-fold CV using AUC.

library(naivebayes)

numeric_cols <- train %>% select(where(is.numeric)) %>% select(-SK_ID_CURR, -TARGET) %>% names()
model_data <- train %>% select(TARGET, all_of(numeric_cols)) %>% drop_na()

set.seed(42)
folds <- createFolds(model_data$TARGET, k = 3, list = TRUE)

ext_source_cols <- c("EXT_SOURCE_1", "EXT_SOURCE_2", "EXT_SOURCE_3")
ext_source_cols <- ext_source_cols[ext_source_cols %in% names(model_data)]
combined_cols <- unique(c(demo_cols, ext_source_cols))

cv_results <- tibble(Model = character(), Fold = integer(), AUC = double())

for (i in seq_along(folds)) {
  idx <- folds[[i]]
  tr <- model_data[-idx, ]
  te <- model_data[idx, ]

  fit_lr <- glm(reformulate(demo_cols, "TARGET"), data = tr, family = binomial)
  pr_lr <- predict(fit_lr, newdata = te, type = "response")
  cv_results <- cv_results %>% add_row(Model = "LR_Demo", Fold = i, AUC = calc_auc(te$TARGET, pr_lr))

  tr_nb <- tr %>% mutate(TARGET = factor(TARGET, levels = c(0, 1)))
  fit_nb <- naive_bayes(reformulate(combined_cols, "TARGET"), data = tr_nb)
  pr_nb <- predict(fit_nb, newdata = te, type = "prob")[, "1"]
  cv_results <- cv_results %>% add_row(Model = "NaiveBayes", Fold = i, AUC = calc_auc(te$TARGET, pr_nb))

  tr_rf <- tr %>% mutate(TARGET = factor(TARGET, levels = c(0, 1)))
  fit_rf <- randomForest(reformulate(combined_cols, "TARGET"), data = tr_rf, ntree = 100)
  pr_rf <- predict(fit_rf, newdata = te, type = "prob")[, "1"]
  cv_results <- cv_results %>% add_row(Model = "RandomForest", Fold = i, AUC = calc_auc(te$TARGET, pr_rf))
}

cv_lr_auc <- function(df) {
  flds <- createFolds(df$TARGET, k = 3, list = TRUE)

    calc_auc(te$TARGET, pr)
  })
  mean(aucs)
}

4. Hyperparameter Tuning

Use randomized search (~20 iterations) on a 5k subsample with 3-fold CV for Random Forest.

set.seed(42)
train_up <- up

tune_df <- train_up %>%
  select(TARGET, all_of(combined_cols)) %>%
  impute_numeric_median() %>%
  sample_n(min(5000, n()))

flds <- createFolds(tune_df$TARGET, k = 3, list = TRUE)

param_grid <- tibble(
  mtry = sample(1:max(1, length(combined_cols)), 20, replace = TRUE),
  ntree = sample(c(100, 150, 200, 300), 20, replace = TRUE),
  nodesize = sample(c(1, 5, 10, 20), 20, replace = TRUE)
)

tuning_results <- tibble(iter = integer(), mtry = integer(), ntree = integer(), nodesize = integer(), Mean_AUC = double())

for (j in seq_len(nrow(param_grid))) {
  p <- param_grid[j, ]
  aucs <- map_dbl(flds, function(idx) {
    tr <- tune_df[-idx, ] %>% mutate(TARGET = factor(TARGET, levels = c(0, 1)))
    te <- tune_df[idx, ]
    fit <- randomForest(reformulate(combined_cols, "TARGET"), data = tr, mtry = p$mtry, ntree = p$ntree, nodesize = p$nodesize)
    pr <- predict(fit, newdata = te, type = "prob")[, "1"]
    calc_auc(te$TARGET, pr)
  })

  tuning_results <- tuning_results %>%
    add_row(iter = j, mtry = p$mtry, ntree = p$ntree, nodesize = p$nodesize, Mean_AUC = mean(aucs, na.rm = TRUE))
}

Use aggregated supplementary features and compare with/without them.

supp_prefix <- c("BUREAU_", "PREV_APP_", "INSTALL_")

compare_rf_auc <- function(feature_set) {
  df <- train %>%
    select(TARGET, all_of(feature_set)) %>%
    impute_numeric_median()

  flds <- createFolds(df$TARGET, k = 3, list = TRUE)

  aucs <- map_dbl(flds, function(idx) {
    tr <- df[-idx, ] %>% mutate(TARGET = factor(TARGET, levels = c(0, 1)))
    te <- df[idx, ]
    fit <- randomForest(
      reformulate(feature_set, "TARGET"),
      data = tr,
      ntree = best_params$ntree,
      mtry = min(best_params$mtry, length(feature_set)),
      nodesize = best_params$nodesize
    )
    pr <- predict(fit, newdata = te, type = "prob")[, "1"]
    calc_auc(te$TARGET, pr)
  })

  mean(aucs)
}

auc_base <- compare_rf_auc(base_features)
auc_enriched <- compare_rf_auc(enriched_features)

supp_results <- tibble(
